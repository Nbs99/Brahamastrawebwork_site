async function handleAuth(e, mode) {
    e.preventDefault();
    
    if (mode === 'login') {
        const email = document.getElementById('login-email').value;
        const pass = document.getElementById('login-pass').value;
        
        try {
            if (auth) {
                const cred = await auth.signInWithEmailAndPassword(email, pass);
                State.user = { email: cred.user.email, uid: cred.user.uid };
            } else {
                State.user = { email, uid: 'local_' + Date.now() };
            }
            saveState();
            closeModal('auth-modal');
            updateUI();
            showCustomAlert('Welcome Back!', `Logged in as ${email}`, 'success');
            showAchievement('Welcome Back', 'You returned to Brahmastra!', '👋');
        } catch(err) {
            if (!auth) {
                State.user = { email, uid: 'local_' + Date.now() };
                saveState();
                closeModal('auth-modal');
                updateUI();
                showCustomAlert('Login Successful', `Welcome ${email}!`, 'success');
            } else {
                showCustomAlert('Login Failed', err.message, 'error');
            }
        }
    } else {
        const username = document.getElementById('signup-user').value;
        const email = document.getElementById('signup-email').value;
        const pass = document.getElementById('signup-pass').value;
        
        try {
            if (auth) {
                const cred = await auth.createUserWithEmailAndPassword(email, pass);
                State.user = { email, uid: cred.user.uid, username };
                if (db) await db.collection('users').doc(cred.user.uid).set({ email, username, wallet: 0, createdAt: firebase.firestore.FieldValue.serverTimestamp() });
            } else {
                State.user = { email, uid: 'local_' + Date.now(), username };
            }
            State.wallet = 50;
            State.loyaltyPoints = 100;
            saveState();
            closeModal('auth-modal');
            updateUI();
            showCustomAlert('Account Created!', '₹50 added to your wallet!', 'success');
            showAchievement('New Warrior', 'Joined Brahmastra! +₹50', '⚔️');
            
            setTimeout(() => openModal('welcome-bonus-modal'), 1000);
        } catch(err) {
            if (!auth) {
                State.user = { email, uid: 'local_' + Date.now(), username };
                State.wallet = 50;
                State.loyaltyPoints = 100;
                saveState();
                closeModal('auth-modal');
                updateUI();
                showCustomAlert('Account Created!', '₹50 added to your wallet!', 'success');
                setTimeout(() => openModal('welcome-bonus-modal'), 1000);
            } else {
                showCustomAlert('Signup Failed', err.message, 'error');
            }
        }
    }
}

// ============ CART FUNCTIONS ============
function addToCart(gameId) {
    const game = GAMES.find(g => g.id === gameId);
    if (!game) return;
    
    if (State.cart.includes(gameId)) {
        showCustomAlert('Already in Cart', `${game.name} is already in your cart!`, 'warning');
        return;
    }
    
    State.cart.push(gameId);
    saveState();
    updateUI();
    showCustomAlert('Added to Cart', `${game.name} - ${formatPrice(game.price)}`, 'success');
    showAchievement('Cart Updated', `${game.name} added!`, '🛒');
}

function removeFromCart(gameId) {
    State.cart = State.cart.filter(id => id !== gameId);
    saveState();
    updateUI();
    renderCart();
}

function payWithWallet(amount) {
    if (State.wallet < amount) {
        showCustomAlert('Insufficient Balance', `You need ${formatPrice(amount - State.wallet)} more in your wallet.`, 'error');
        return;
    }
    
    State.wallet -= amount;
    const orderId = generateOrderId();
    const games = State.cart.map(id => GAMES.find(g => g.id === id)).filter(Boolean);
    
    games.forEach(g => {
        State.orders.unshift({
            id: orderId,
            game: g.name,
            price: g.price,
            date: new Date().toLocaleDateString(),
            status: 'Delivered',
            img: g.img
        });
    });
    
    State.cart = [];
    State.loyaltyPoints += Math.floor(amount / 10);
    saveState();
    updateUI();
    closeModal('game-modal');
    
    showCustomAlert('Payment Successful!', `Order ${orderId} confirmed!`, 'success');
    showAchievement('Purchase Complete', `+${Math.floor(amount / 10)} XP earned!`, '💰');
    triggerAirdrop();
}

// ============ CATALOG FUNCTIONS ============
function getFilteredGames() {
    let games = [...GAMES];
    
    // Search
    if (State.searchQuery) {
        const q = State.searchQuery.toLowerCase();
        games = games.filter(g => 
            g.name.toLowerCase().includes(q) || 
            g.genre.some(genre => genre.toLowerCase().includes(q)) ||
            g.platform.toLowerCase().includes(q) ||
            g.tags.some(tag => tag.toLowerCase().includes(q))
        );
    }
    
    // Platform filter
    if (State.platformFilter !== 'All') {
        games = games.filter(g => g.platform === State.platformFilter);
    }
    
    // Sort
    switch (State.sortMode) {
        case 'price-asc': games.sort((a, b) => a.price - b.price); break;
        case 'price-desc': games.sort((a, b) => b.price - a.price); break;
        case 'rating-desc': games.sort((a, b) => b.rating - a.rating); break;
    }
    
    return games;
}

function sendAIMessage() {
    const input = document.getElementById('ai-bot-input');

    if (!message) return;
    
    const container = document.getElementById('ai-bot-messages');
    
    // User message
    container.innerHTML += `
        <div class="flex justify-end">
            <div class="bg-brahma-purple/30 border border-brahma-purple/50 p-3 rounded-xl rounded-tr-none max-w-[85%]">
                <p class="text-white text-[11px] leading-relaxed uppercase">${message}</p>

    const typing = document.getElementById('ai-typing-indicator');
    typing.classList.remove('hidden');
    typing.classList.add('flex');
    
    // Generate response
    setTimeout(() => {
        typing.classList.add('hidden');
        typing.classList.remove('flex');
        
        let response = AI_RESPONSES.default();
        const msg = message.toLowerCase();
        
        if (msg.includes('requirement') || msg.includes('spec') || msg.includes('pc')) {
            const gameName = message.replace(/requirements?|specs?|pc/gi, '').trim();

            response = game ? AI_RESPONSES.requirements(game.name) : AI_RESPONSES.requirements('Game');
        } else if (msg.includes('track') || msg.includes('order')) {
            response = AI_RESPONSES.track();
        } else if (msg.includes('refund') || msg.includes('cancel')) {
            response = AI_RESPONSES.refund();
        } else if (msg.includes('payment') || msg.includes('pay') || msg.includes('upi')) {
            response = AI_RESPONSES.payment();
        } else if (msg.includes('shared') || msg.includes('access')) {
            response = AI_RESPONSES.shared();
        } else if (msg.includes('contact') || msg.includes('support') || msg.includes('whatsapp')) {
            response = `📞 Contact Support:\n\n📱 WhatsApp: +91 9313339081\n📧 Email: support@brahmastra.store\n\nWe respond within 5 minutes!`;
        } else if (msg.includes('hello') || msg.includes('hi') || msg.includes('hey')) {
            response = `👋 Hello Gamer! How can I help you today?\n\nTry asking about:\n• Game requirements\n• Order tracking\n• Payment methods\n• Shared access`;
        } else if (msg.includes('thank')) {
            response = `😊 You're welcome! Happy gaming! 🎮`;
        }
        
        container.innerHTML += `
            <div class="flex justify-start">
                <div class="bg-purple-900/20 border border-purple-500/30 p-3 rounded-xl rounded-tl-none max-w-[85%]">
                    <p class="text-white text-[11px] leading-relaxed uppercase" style="white-space: pre-line;">${response}</p>
                </div>
            </div>
        `;
        
        container.scrollTop = container.scrollHeight;
    }, 1500);
}

function startJarvisVoice() {
    if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
        showCustomAlert('Not Supported', 'Voice search is not supported in this browser', 'warning');
        return;
    }
    
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    const recognition = new SpeechRecognition();
    recognition.lang = 'en-US';
    recognition.continuous = false;
    
    showCustomAlert('Listening...', 'Speak the game name...', 'info');
    
    recognition.onresult = (event) => {
        const text = event.results[0][0].transcript;
        document.getElementById('game-search').value = text;
        handleSearch(text);
        closeCustomAlert();
    };
    
    recognition.onerror = () => {
        showCustomAlert('Error', 'Could not understand. Try again.', 'error');
    };
    
    recognition.start();
}

function changeCarouselCategory(cat) {
    document.querySelectorAll('.carousel-cat-btn').forEach(btn => {
        btn.className = 'carousel-cat-btn px-5 py-2 border border-white/20 text-white hover:border-brahma-cyan font-bold uppercase rounded-lg text-xs transition-colors tracking-widest';
    });
    event.target.className = 'carousel-cat-btn active px-5 py-2 bg-brahma-cyan text-black font-bold uppercase rounded-lg text-xs tracking-widest';
    
    if (cat === 'All') {
        carouselGames = GAMES.slice(0, 9);
    } else {
        carouselGames = GAMES.filter(g => g.genre.includes(cat)).slice(0, 9);
    }
    
    carouselIndex = 0;
    renderCarousel();
}

function renderCarousel() {
    const container = document.getElementById('carousel-3d');
    container.innerHTML = '';
    
    const count = carouselGames.length;
    const angle = 360 / count;
    const radius = 350;
    
    carouselGames.forEach((game, i) => {
        const card = document.createElement('div');
        card.className = 'carousel-card';
        card.onclick = () => openGameModal(game.id);
        
        const currentAngle = angle * i;
        card.style.transform = `rotateY(${currentAngle}deg) translateZ(${radius}px)`;
        
        card.innerHTML = `
            <img src="${game.img}" alt="${game.name}">
            <div class="card-overlay">
                <h4 class="text-white font-bold text-sm uppercase">${game.name}</h4>
                <p class="text-brahma-cyan font-bold text-xs mt-1">${formatPrice(game.price)}</p>
            </div>
        `;
        
        container.appendChild(card);
    });
    
    updateCarousel();
}

function rotateCarousel(dir) {
    carouselIndex += dir === 'next' ? 1 : -1;
    updateCarousel();
}

function updateCarousel() {
    const container = document.getElementById('carousel-3d');
    const cards = container.querySelectorAll('.carousel-card');
    const count = cards.length;
    const angle = 360 / count;
    
    container.style.transform = `rotateY(${-angle * carouselIndex}deg)`;
}

// Auto-rotate carousel
setInterval(() => rotateCarousel('next'), 5000);

// ============ FLASH SALE TIMER ============
function startFlashSaleTimer() {
    let hours = 2, minutes = 15, seconds = 30;
    
    setInterval(() => {
        seconds--;
        if (seconds < 0) { seconds = 59; minutes--; }
        if (minutes < 0) { minutes = 59; hours--; }
        if (hours < 0) { hours = 23; minutes = 59; seconds = 59; }
        
        const timer = document.getElementById('flash-sale-timer');
        if (timer) {
            timer.textContent = `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
        }
    }, 1000);
}

// ============ PARALLAX ENGINE ============
const ParallaxEngine = {
    rafId: null,
    scrollY: 0,
    windowHeight: 0,
    elements: [],
    
    init() {
        this.windowHeight = window.innerHeight;
        this.cacheElements();
        this.bindEvents();
        this.tick();
    },
    
    cacheElements() {
        this.elements = [];
        document.querySelectorAll('.parallax-wrap').forEach(wrap => {
            const card = wrap.querySelector('.parallax-card');
            if (card) this.elements.push({ type: 'card', wrap, card, depthImg: wrap.querySelector('.p-depth-img'), depthContent: wrap.querySelector('.p-depth-content'), depthBadge: wrap.querySelector('.p-depth-badge'), rect: null });
        });
        document.querySelectorAll('.logo-3d-container').forEach(logo => {
            const inner = logo.querySelector('.logo-3d-inner');
            if (inner) this.elements.push({ type: 'logo', el: logo, inner, rect: null });
        });
        document.querySelectorAll('.elite-container').forEach(container => {
            const card = container.querySelector('.elite-card');
            if (card) {
                this.elements.push({ type: 'elite', card, content: card.querySelector('.elite-content'), badge: card.querySelector('.elite-badge'), rect: null });
                container.addEventListener('mousemove', (e) => {
                    const rect = card.getBoundingClientRect();
                    card.style.setProperty('--mouse-x', ((e.clientX - rect.left) / rect.width * 100) + '%');
                    card.style.setProperty('--mouse-y', ((e.clientY - rect.top) / rect.height * 100) + '%');
                    card.style.transform = `perspective(1000px) rotateX(${((e.clientY - rect.top) / rect.height - 0.5) * -8}deg) rotateY(${((e.clientX - rect.left) / rect.width - 0.5) * 8}deg) scale3d(1.02,1.02,1.02)`;
                });
                container.addEventListener('mouseleave', () => { card.style.transform = ''; });
            }
        });
        document.querySelectorAll('#games-grid > div').forEach(item => {
            item.addEventListener('mousemove', (e) => {
                const rect = item.getBoundingClientRect();
                item.style.transform = `perspective(800px) rotateX(${((e.clientY - rect.top) / rect.height - 0.5) * -10}deg) rotateY(${((e.clientX - rect.left) / rect.width - 0.5) * 10}deg) translateZ(15px)`;
            });
            item.addEventListener('mouseleave', () => { item.style.transform = ''; });
        });
    },
    
    bindEvents() {
        window.addEventListener('scroll', () => { this.scrollY = window.pageYOffset; }, { passive: true });
        window.addEventListener('resize', () => { this.windowHeight = window.innerHeight; }, { passive: true });
    },
    
    tick() {
        this.elements.forEach(el => {
            el.rect = el.wrap?.getBoundingClientRect() || el.el?.getBoundingClientRect() || el.card?.getBoundingClientRect();
            if (!el.rect || el.rect.bottom < -100 || el.rect.top > this.windowHeight + 100) return;
            
            const dist = (el.rect.top + el.rect.height / 2 - this.windowHeight / 2) / this.windowHeight;
            
            if (el.type === 'card') {
                el.card.style.transform = `rotateY(${dist * -15}deg) rotateX(${dist * 8}deg) translateZ(${Math.abs(dist) * -30}px)`;
                if (el.depthImg) el.depthImg.style.transform = `translateZ(-40px) scale(1.12) translateY(${dist * -20}px)`;
                if (el.depthContent) el.depthContent.style.transform = `translateZ(-15px) translateY(${dist * -8}px)`;
                if (el.depthBadge) el.depthBadge.style.transform = `translateZ(25px) translateY(${dist * -5}px)`;
            } else if (el.type === 'logo') {
                el.inner.style.transform = `rotateY(${dist * -10}deg) translateZ(${Math.abs(dist) * -20}px) scale(${Math.max(0.8, 1 + Math.abs(dist) * -0.1)})`;
            }
        });
        this.rafId = requestAnimationFrame(() => this.tick());
    }
};

// ============ SCROLL REVEAL ============
function initScrollReveal() {
    const observer = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                entry.target.style.opacity = '1';
                entry.target.style.transform = 'translateY(0)';
            }
        });
    }, { threshold: 0.1, rootMargin: '0px 0px -50px 0px' });
    
    document.querySelectorAll('section, main > div').forEach((el, i) => {
        el.style.opacity = '0';
        el.style.transform = 'translateY(30px)';
        el.style.transition = `opacity 0.6s ease-out ${i * 0.05}s, transform 0.6s ease-out ${i * 0.05}s`;
        observer.observe(el);
    });
}

// ============ SCROLL PROGRESS BAR ============
function initScrollProgress() {
    const bar = document.createElement('div');
    bar.id = 'scroll-progress-bar';
    bar.style.cssText = 'position:fixed;top:32px;left:0;height:2px;background:linear-gradient(90deg,#06b6d4,#7c3aed);z-index:999;transition:width 0.1s linear;box-shadow:0 0 10px rgba(6,182,212,0.5);';
    document.body.appendChild(bar);
    
    window.addEventListener('scroll', () => {
        const progress = (window.pageYOffset / (document.documentElement.scrollHeight - window.innerHeight)) * 100;
        bar.style.width = progress + '%';
    }, { passive: true });
}

   REAL-TIME SYNC ENGINE
   Admin Panel → Website Live Updates

const RealtimeSync = {
    channel: null,
    lastTimestamps: {},
    
    init() {
        // BroadcastChannel for same-browser tabs
        try {
            this.channel = new BroadcastChannel('brahmastra_sync');
            this.channel.onmessage = (e) => this.handleMessage(e.data);
            console.log('📡 RealtimeSync: BroadcastChannel connected');
        } catch(e) {
            console.warn('BroadcastChannel not supported, using polling');
        }
        
        // Load initial overrides
        this.loadAllOverrides();
        
        // Polling fallback (every 2 seconds)
        setInterval(() => this.pollForChanges(), 2000);
        
        // Listen for storage events from other tabs
        window.addEventListener('storage', (e) => {
            if (e.key && e.key.startsWith('brahmastra_')) {
                this.loadAllOverrides();
            }
        });
    },
    
    handleMessage(data) {
        if (!data || !data.type) return;
        console.log('📡 Realtime update:', data.type);
        
        switch(data.type) {
            case 'games':
                this.applyGames(data.data);
                break;
            case 'deals':
                this.applyDeals(data.data);
                break;
            case 'banner':
                this.applyBanner(data.data);
                break;
            case 'broadcast':
                this.applyBroadcast(data.data);
                break;
            case 'reviews':
                this.applyReviews(data.data);
                break;
            case 'wallet':
                if (State.user && data.data.email === State.user.email) {
                    State.wallet = data.data.wallet;
                    saveState();
                    updateUI();
                    showCustomAlert('Wallet Updated!', `New balance: ${formatPrice(State.wallet)}`, 'info');
                }
                break;
            case 'secret_codes':
                // Codes update in real-time via applySecretAccess
                break;
        }
    },
    
    pollForChanges() {
        const keys = ['brahmastra_games_ts', 'brahmastra_deals_ts', 'brahmastra_banner_ts', 'brahmastra_broadcast_ts', 'brahmastra_reviews_ts'];
        
        keys.forEach(key => {
            const ts = localStorage.getItem(key);
            const type = key.replace('_ts', '').replace('brahmastra_', '');
            
            if (ts && this.lastTimestamps[type] !== ts) {
                this.lastTimestamps[type] = ts;
                this.loadOverride(type);
            }
        });
    },
    
    loadAllOverrides() {
        this.loadOverride('games');
        this.loadOverride('deals');
        this.loadOverride('banner');
        this.loadOverride('broadcast');
        this.loadOverride('reviews');
    },
    
    loadOverride(type) {
        try {
            const data = JSON.parse(localStorage.getItem('brahmastra_' + type));
            if (!data) return;
            
            switch(type) {
                case 'games': this.applyGames(data); break;
                case 'deals': this.applyDeals(data); break;
                case 'banner': this.applyBanner(data); break;
                case 'broadcast': this.applyBroadcast(data); break;
                case 'reviews': this.applyReviews(data); break;
            }
        } catch(e) {}
    },
    
    applyGames(games) {
        if (!Array.isArray(games) || games.length === 0) return;
        
        // Override the GAMES array reference
        window.GAMES.length = 0;
        games.forEach(g => window.GAMES.push(g));
        
        renderCatalog();
        renderRecentlyViewed();
        initCarousel();
        initCube();
        console.log('🔄 Games updated in real-time:', games.length, 'games');
    },
    
    applyDeals(deals) {
        if (!deals) return;
        
        if (deals.d1) {
            const el1 = document.getElementById('deal-name-1');
            const pr1 = document.getElementById('deal-price-1');
            if (el1) el1.textContent = deals.d1.name;
            if (pr1) pr1.textContent = '₹' + (deals.d1.price || 0).toLocaleString('en-IN');
        }
        if (deals.d2) {
            const el2 = document.getElementById('deal-name-2');
            const pr2 = document.getElementById('deal-price-2');
            if (el2) el2.textContent = deals.d2.name;
            if (pr2) pr2.textContent = '₹' + (deals.d2.price || 0).toLocaleString('en-IN');
        }
        if (deals.d3) {
            const el3 = document.getElementById('deal-name-3');
            const pr3 = document.getElementById('deal-price-3');
            if (el3) el3.textContent = deals.d3.name;
            if (pr3) pr3.textContent = '₹' + (deals.d3.price || 0).toLocaleString('en-IN');
        }
        console.log('🔄 Deals updated in real-time');
    },
    
    applyBanner(items) {
        if (!Array.isArray(items) || items.length === 0) return;
        
        const container = document.getElementById('top-banner-content');
        if (!container) return;
        
        // Duplicate for seamless loop
        const allItems = [...items, ...items];
        container.innerHTML = allItems.map(item => 
            `<span>${item}</span>`
        ).join('');
        console.log('🔄 Banner updated in real-time');
    },
    
    applyBroadcast(data) {
        if (!data) return;
        
        if (data.active && data.text) {
            showGlobalBroadcast(data.text);
        } else {
            hideGlobalBanner();
        }
        console.log('🔄 Broadcast updated in real-time');
    },
    
    applyReviews(reviews) {
        if (!Array.isArray(reviews) || reviews.length === 0) return;
        
        // Update the global REVIEWS array
        window.REVIEWS.length = 0;
        reviews.forEach(r => window.REVIEWS.push(r));
        
        // Re-render review slider
        initReviewSlider();
        console.log('🔄 Reviews updated in real-time:', reviews.length, 'reviews');
    }
};

// Override applySecretAccess to use dynamic codes
const _originalApplySecretAccess = applySecretAccess;
applySecretAccess = function() {
    const code = document.getElementById('secret-access-input').value.trim().toUpperCase();
    
    // Check dynamic codes from admin
    try {
        const dynamicCodes = JSON.parse(localStorage.getItem('brahmastra_secret_codes') || '{}');
        if (dynamicCodes[code]) {
            const data = dynamicCodes[code];
            if (data.reward > 0) State.wallet += data.reward;
            if (data.xp > 0) State.loyaltyPoints += data.xp;
            saveState();
            updateUI();
            showCustomAlert('Access Granted!', `₹${data.reward} + ${data.xp} XP added!`, 'success');
            showAchievement('Secret Unlocked', 'You found a hidden code!', '🔓');
            document.getElementById('secret-access-input').value = '';
            return;
        }
    } catch(e) {}
    
    // Fallback to hardcoded codes
    _originalApplySecretAccess();
};
